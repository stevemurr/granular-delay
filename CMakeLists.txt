cmake_minimum_required(VERSION 3.22)
project(GranularDelay VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GRANULAR_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(MSVC)
  add_compile_options(/W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  if(GRANULAR_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()

add_library(granular_dsp STATIC src/CloudEngine.cpp src/SpectralEngine.cpp src/Processor.cpp src/Reverb.cpp)
target_include_directories(granular_dsp PUBLIC src)
set_target_properties(granular_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)

add_library(GranularDelay MODULE src/Plugin.cpp)
target_link_libraries(GranularDelay PRIVATE granular_dsp)
target_include_directories(GranularDelay PRIVATE third_party/clap/include)
set_target_properties(GranularDelay PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  add_library(granular_editor STATIC src/Editor.mm)
  target_compile_options(granular_editor PRIVATE -fobjc-arc)
  target_include_directories(granular_editor PUBLIC src)
  target_link_libraries(granular_editor PUBLIC "-framework AppKit" "-framework QuartzCore")
  target_link_libraries(GranularDelay PRIVATE granular_editor)
  add_executable(EditorPreview MACOSX_BUNDLE tools/EditorPreview.mm src/Plugin.cpp)
  target_compile_options(EditorPreview PRIVATE -fobjc-arc)
  target_include_directories(EditorPreview PRIVATE third_party/clap/include)
  target_link_libraries(EditorPreview PRIVATE granular_editor granular_dsp)
  set_target_properties(EditorPreview PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER com.stevemurr.granular-delay.preview)
  set_target_properties(granular_editor PROPERTIES CXX_VISIBILITY_PRESET hidden OBJCXX_VISIBILITY_PRESET hidden)
  set_target_properties(GranularDelay PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION clap
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist.in")
  add_custom_command(TARGET GranularDelay POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:GranularDelay>/Contents/Resources"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES.md"
      "$<TARGET_BUNDLE_DIR:GranularDelay>/Contents/Resources/THIRD_PARTY_NOTICES.md"
    COMMAND codesign --force --sign - "$<TARGET_BUNDLE_DIR:GranularDelay>"
    COMMENT "Ad-hoc signing the local CLAP bundle")
  install(TARGETS GranularDelay LIBRARY DESTINATION . BUNDLE DESTINATION .)
else()
  set_target_properties(GranularDelay PROPERTIES SUFFIX .clap)
  install(TARGETS GranularDelay LIBRARY DESTINATION . RUNTIME DESTINATION .)
  install(FILES THIRD_PARTY_NOTICES.md DESTINATION .)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(engine_tests tests/EngineTests.cpp)
  target_link_libraries(engine_tests PRIVATE granular_dsp)
  add_test(NAME engines COMMAND engine_tests)
  add_executable(processor_tests tests/ProcessorTests.cpp)
  target_link_libraries(processor_tests PRIVATE granular_dsp)
  add_test(NAME processor COMMAND processor_tests)
  add_executable(plugin_tests tests/PluginTests.cpp src/Plugin.cpp)
  target_include_directories(plugin_tests PRIVATE third_party/clap/include)
  target_link_libraries(plugin_tests PRIVATE granular_dsp)
  if(APPLE)
    target_link_libraries(plugin_tests PRIVATE granular_editor)
  endif()
  add_test(NAME plugin COMMAND plugin_tests)
endif()
add_executable(render_demo tools/RenderDemo.cpp)
target_link_libraries(render_demo PRIVATE granular_dsp)
if(APPLE)
  add_executable(editor_snapshot tools/EditorSnapshot.mm)
  target_compile_options(editor_snapshot PRIVATE -fobjc-arc)
  target_link_libraries(editor_snapshot PRIVATE granular_editor)
endif()
